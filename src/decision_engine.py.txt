def maintenance_recommendation(
    failure_probability,
    estimated_cost
):
    if failure_probability >= 0.75:
        recommendation = "Immediate maintenance should be considered."
    elif failure_probability >= 0.50:
        recommendation = "Maintenance should be planned soon."
    else:
        recommendation = "Continue monitoring equipment."

    return {
        "failure_probability": round(failure_probability, 3),
        "estimated_cost": round(estimated_cost, 2),
        "recommendation": recommendation
    }


# Example decision
result = maintenance_recommendation(
    failure_probability=0.75,
    estimated_cost=6200
)

print("Maintenance Decision:")
print(result)
